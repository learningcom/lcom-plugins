CREATE TABLE IF NOT EXISTS support.fact_case
(
	case_id VARCHAR(300) NOT NULL  ENCODE lzo
	,account_id VARCHAR(300) NOT NULL  ENCODE lzo
	,sfdc_account_id VARCHAR(300) NOT NULL  ENCODE lzo
	,case_number VARCHAR(90) NOT NULL  ENCODE lzo
	,case_owner_email VARCHAR(400) NOT NULL  ENCODE lzo
	,closed_date TIMESTAMP WITHOUT TIME ZONE NOT NULL  ENCODE az64
	,contact_id VARCHAR(300) NOT NULL  ENCODE lzo
	,created_date TIMESTAMP WITHOUT TIME ZONE NOT NULL  ENCODE RAW
	,description VARCHAR(65535) NOT NULL  ENCODE lzo
	,is_closed BOOLEAN NOT NULL  ENCODE RAW
	,is_escalated BOOLEAN NOT NULL  ENCODE RAW
	,last_modified_date TIMESTAMP WITHOUT TIME ZONE NOT NULL  ENCODE az64
	,origin VARCHAR(1000) NOT NULL  ENCODE lzo
	,owner_id VARCHAR(300) NOT NULL  ENCODE lzo
	,platform_name VARCHAR(300) NOT NULL  ENCODE lzo
	,case_priority VARCHAR(1000) NOT NULL  ENCODE lzo
	,support_type VARCHAR(240) NOT NULL  ENCODE lzo
	,solution VARCHAR(5000) NOT NULL  ENCODE lzo
	,status VARCHAR(100) NOT NULL  ENCODE lzo
	,subject VARCHAR(1000) NOT NULL  ENCODE lzo
	,case_type VARCHAR(100) NOT NULL  ENCODE lzo
	,loaddate TIMESTAMP WITHOUT TIME ZONE NOT NULL  ENCODE az64
	,PRIMARY KEY (case_id)
)
DISTSTYLE KEY
 DISTKEY (account_id)
 SORTKEY (
	created_date
	)
;
ALTER TABLE support.fact_case owner to svcfivetran;
COMMENT ON TABLE support.fact_case IS 'Fact Case (Salesforce Case table)';
