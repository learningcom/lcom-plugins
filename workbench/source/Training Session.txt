	topic_id
	)

COMMENT ON COLUMN content_delivery_usage.dim_training_session_topic.topic_id IS 'Unique topic identifier (MD5 hash of topic name). Primary key.';

COMMENT ON COLUMN content_delivery_usage.dim_training_session_topic.loaddate IS 'The datetime the record was loaded in PST';

	topic_id
	)

COMMENT ON TABLE content_delivery_usage.dim_training_session_topic_session IS 'Mapping table between training sessions and topics. Each record links a training session to a topic, with load timestamp. Used for analyzing topic coverage across sessions.';
